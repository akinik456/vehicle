const SHEET_NAME = "LynraFleet Mail";
const SUBJECT = "Araçlarınızı ek GPS cihazı olmadan takip edin";

// İlk çalıştırmada 5 yap.
// Her şey düzgünse 20-30 yapabiliriz.
const MAX_PER_RUN = 50;


// =====================================================
// TOPLU GÖNDERİM
// =====================================================

function sendLynraFleetBulk() {

  const sheet = SpreadsheetApp.getActiveSpreadsheet()
    .getSheetByName(SHEET_NAME);

  if (!sheet) {
    throw new Error("Sayfa bulunamadı: " + SHEET_NAME);
  }

  const lastRow = sheet.getLastRow();

  if (lastRow < 2) {
    Logger.log("Gönderilecek firma yok.");
    return;
  }

  const rows = sheet
    .getRange(2, 1, lastRow - 1, 3)
    .getValues();

  let sentCount = 0;

  for (let i = 0; i < rows.length; i++) {

    const rowNumber = i + 2;

    const firma = String(rows[i][0] || "").trim();
    const email = String(rows[i][1] || "").trim();
    const gonderildi = rows[i][2];

    // Daha önce gönderilmiş
    if (gonderildi) {
      continue;
    }

    // Eksik veri
    if (!firma || !email) {
      Logger.log(
        "ATLANDI => row=" + rowNumber +
        " firma=" + firma +
        " email=" + email
      );
      continue;
    }

    // Bu çalıştırmadaki maksimum sayı
    if (sentCount >= MAX_PER_RUN) {
      Logger.log("MAX_PER_RUN sınırına ulaşıldı.");
      break;
    }

    // Gmail günlük kota kontrolü
    const remainingQuota =
      MailApp.getRemainingDailyQuota();

    if (remainingQuota <= 0) {
      Logger.log("Gmail günlük gönderim kotası bitti.");
      break;
    }

    try {

      const html = buildLynraFleetHtml(firma);

      GmailApp.sendEmail(
        email,
        SUBJECT,
        "LynraFleet araç takip çözümü hakkında bilgi.",
        {
          htmlBody: html,
          name: "LynraFleet"
        }
      );

      // SADECE başarılı gönderimde işaretle
      sheet
        .getRange(rowNumber, 3)
        .setValue(new Date());

      SpreadsheetApp.flush();

      sentCount++;

      Logger.log(
        "GÖNDERİLDİ => " +
        firma +
        " => " +
        email
      );

      // Çok hızlı arka arkaya bindirmeyelim
      Utilities.sleep(1000);

    } catch (e) {

      Logger.log(
        "HATA => row=" +
        rowNumber +
        " firma=" +
        firma +
        " email=" +
        email +
        " error=" +
        e
      );

      // C sütununu doldurmuyoruz.
      // Böylece sonraki çalıştırmada tekrar denenebilir.
    }
  }

  Logger.log(
    "BİTTİ => bu çalıştırmada gönderilen=" +
    sentCount
  );
}
